  function isActiveClientOrg(ctx: SurveyDeliveryContext): boolean {
    const salesforceOrgId = ctx.salesforceOrgId?.trim();
    const clientOrgId = ctx.clientOrgId?.trim().toLowerCase();
    if (!salesforceOrgId && !clientOrgId) return false;
    if (salesforceOrgId) {
      const foundByOrgId = db.select({...}).from(organizations)
        .where(and(eq(organizations.salesforceOrgId, salesforceOrgId), eq(organizations.active, true)))
        ...
      if (foundByOrgId) return true;           // <-- primary path
    }
    if (!clientOrgId) return false;
    return db.select({...}).from(organizations)
      .where(and(eq(organizations.subdomain, clientOrgId), eq(organizations.active, true)))
      ...                                       // <-- fallback path (AC8, dev/stg only in practice)
  }
