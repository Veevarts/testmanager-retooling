## Slug/path validation (client-side)

lib/subjectForm.ts:92-107:
  export function subjectFormValidationError(form: SubjectFormState): string | null {
    if (!form.name.trim()) return "Enter a subject name.";
    if (!form.slug.trim()) return "Enter a subject slug.";
    if (!/^[a-z0-9_-]+$/i.test(form.slug.trim())) {
      return "The slug can contain only letters, numbers, dashes, and underscores.";
    }
    ...
  }

Slug normalizer (lib/subjectForm.ts:46-54):
  export function normalizeSubjectSlug(value: string): string {
    return value
      .normalize("NFD")
      .replace(/[̀-ͯ]/g, "")
      .toLowerCase()
      .replace(/[^a-z0-9_-]+/g, "-")
      .replace(/-+/g, "-")
      .replace(/^[-_]+|[-_]+$/g, "");
  }
