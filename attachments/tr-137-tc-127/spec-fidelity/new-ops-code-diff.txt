+/**
+ * Deploy a Lightning **Help Menu** custom section (`CustomHelpMenuSection`) via
+ * the synchronous CRUD Metadata API: read the section by `fullName` (may be
+ * absent) → build `{ fullName, masterLabel, customHelpMenuItems }` → UPSERT
+ * (create when absent, update when present) → read-back verify the `masterLabel`
+ * and the section's items landed.
+ *
+ * **Why a dedicated op (vs T6 Apex):** the Help Menu is a metadata component,
+ * NOT data — anonymous Apex cannot create or edit a `CustomHelpMenuSection`.
+ * The synchronous CRUD Metadata API writes then re-reads in one shot, matching
+ * the per-step verification convention.
+ *
+ * **Metadata shape (verified via describeValueType + a real-org deploy):** the
+ * component type is `CustomHelpMenuSection` with `fullName`, `masterLabel`, and
+ * a repeated child `customHelpMenuItems` whose element type
+ * `CustomHelpMenuItem` carries `masterLabel` (the ITEM label — note the field is
+ * NOT called `label`), `linkUrl` (the URL), and a REQUIRED int `sortOrder`. So
+ * the adapter maps each `items[i]` → `{ masterLabel: item.label, linkUrl:
+ * item.url, sortOrder: item.sortOrder ?? (i + 1) }`.
+ *
+ * Verified by read-back; a write that didn't take is `success=false`, never a
+ * throw (rules §6).
+ */
+export interface CustomHelpMenuSectionSpec {
+    readonly op: 'customHelpMenuSection';
+    /** Metadata fullName of the section (e.g. `Veevart_Resources`). */
+    readonly fullName: string;
+    /** Section header label. */
+    readonly masterLabel: string;

+/**
+ * UPSERT custom-label **translation overrides** for one platform language via
+ * the synchronous CRUD Metadata API: `metadata.update('Translations', {
+ * fullName: language, customLabels: [{ name, label }, …] })`.
+ *
+ * **Why no read-merge (the key CRUD semantics, tested live):**
+ * `metadata.update('Translations', …)` is a MERGE/UPSERT keyed by each label's
+ * `name` — it sets ONLY the named labels and leaves every OTHER custom label
+ * untouched. So this op does NOT read+merge the (thousands of) existing labels
+ * before writing; it just calls update with the target labels. (Reading the
+ * full `Translations` component returns EVERY custom label — very large — so the
+ * op reads it back only to VERIFY, then filters to the target `name`s and
+ * asserts each `label` matches; it never logs the whole component.)
+ *
+ * **Set-only (document this):** overrides are UPSERT-only via the API — an
+ * override cannot be CLEARED to blank (`label: ''` is ignored, and there is no
+ * delete), so the op is a set-only idempotent operation, which fits the use
+ * case (installing the museum's label overrides). Re-applying the same labels
+ * re-writes the same values and still verifies.
+ *
+ * Verified by read-back (filtered to the target names); a write that didn't
+ * take is `success=false`, never a throw (rules §6).
+ */
+export interface TranslationsSpec {
+    readonly op: 'translations';
+    /** Platform language of the override; defaults to `en_US`. */
+    readonly language?: string;
+    /** Custom-label overrides to UPSERT (idempotent set). At least one required. */
+    readonly customLabels: readonly TranslationCustomLabelSpec[];
+}

+    private async applyCustomHelpMenuSection(
+        orgId: string,
+        conn: MetadataConn,
+        spec: CustomHelpMenuSectionSpec,
+    ): Promise<MetadataMutationResult> {
+        this.requireNonBlank(
+            orgId,
+            spec.fullName,
+            'customHelpMenuSection fullName',
+        );
+        this.requireNonBlank(
+            orgId,
+            spec.masterLabel,
+            'customHelpMenuSection masterLabel',
+        );
+        const items = spec.items ?? [];
+        if (items.length === 0) {
+            throw new MetadataDeployError(
+                orgId,
+                'MALFORMED_REQUEST',
+                `T3 customHelpMenuSection ${spec.fullName} has an empty items list.`,
+            );
+        }
+        for (const item of items) {
+            this.requireNonBlank(
+                orgId,
+                item?.label,
+                'customHelpMenuSection item label',
+            );
+            this.requireNonBlank(
+                orgId,
+                item?.url,
+                'customHelpMenuSection item url',
+            );
+        }
+        const fullName = spec.fullName.trim();
+        const masterLabel = spec.masterLabel.trim();
+        // Map each spec item onto the verified CustomHelpMenuItem shape: the
+        // item's `label` → `masterLabel`, `url` → `linkUrl`, and a REQUIRED
+        // 1-based `sortOrder` defaulting to the declaration index.
+        const customHelpMenuItems: CustomHelpMenuItem[] = items.map(
+            (item, i): CustomHelpMenuItem => ({
+                masterLabel: item.label,
+                linkUrl: item.url,
+                sortOrder: item.sortOrder ?? i + 1,
+            }),
+        );
+        this.logger.log({
+            action: 't3.customHelpMenuSection',
+            orgId,
+            fullName,
+            itemCount: customHelpMenuItems.length,
+        });
+
+        // 1. Read existing section (may be absent → create; present → update).
+        const before = await this.readOptional<CustomHelpMenuSection>(
+            orgId,
+            conn,
+            'CustomHelpMenuSection',
+            fullName,
+        );
+
+        // 2. Build the full component payload (update is a replace).
+        const payload: CustomHelpMenuSection = {
+            fullName,
+            masterLabel,
+            customHelpMenuItems,
+        };
+
+        // 3. UPSERT via the synchronous CRUD Metadata API.
+        const save = await this.write(
+            orgId,
+            conn,
+            'CustomHelpMenuSection',
+            payload,
+            () =>
+                before
+                    ? conn.metadata.update('CustomHelpMenuSection', payload)
+                    : conn.metadata.create('CustomHelpMenuSection', payload),
+        );
+        const planned = customHelpMenuItems.map((i) => ({
+            subject: i.masterLabel,
+        }));
+        if (coerceBoolean(save?.success) !== true) {
+            return failedWrite(
+                fullName,
+                [{ subject: `${fullName} (masterLabel)` }, ...planned],
+                joinErrors(save) ??
+                    `metadata write on CustomHelpMenuSection ${fullName} reported success=false with no error detail.`,
+                'customHelpMenuSection',
+            );
+        }
+
+        // 4. Verify by read-back: the masterLabel + each item (by label + url)
+        //    landed, and the section carries exactly the requested item count.
+        const after = await this.readOptional<CustomHelpMenuSection>(
+            orgId,
+            conn,
+            'CustomHelpMenuSection',
+            fullName,
+        );
+        const afterItems = asArray<CustomHelpMenuItem>(
+            after?.customHelpMenuItems,
+        );
+        const labelOk =
+            after?.masterLabel === masterLabel &&
+            afterItems.length === customHelpMenuItems.length;
+        const operations: MetadataOpOutcome[] = [
+            {
+                op: 'customHelpMenuSection',
+                subject: `${fullName} (masterLabel)`,
+                applied: labelOk,
+                detail: labelOk
+                    ? null
+                    : `section ${fullName} after write has masterLabel ${JSON.stringify(after?.masterLabel)} and ${afterItems.length} item(s) (expected ${JSON.stringify(masterLabel)} and ${customHelpMenuItems.length})`,
+            },
+            ...customHelpMenuItems.map((item): MetadataOpOutcome => {
+                const found = afterItems.some(
+                    (a) =>
+                        a.masterLabel === item.masterLabel &&
+                        normalizeUrl(a.linkUrl) === normalizeUrl(item.linkUrl),
+                );
+                return {
+                    op: 'customHelpMenuSection',
+                    subject: item.masterLabel,
+                    applied: found,
+                    detail: found
+                        ? null
+                        : `help menu item "${item.masterLabel}" (${item.linkUrl}) is not on section ${fullName} after write`,
+                };
+            }),
+        ];
+        return finish(fullName, operations, this.logger, {
+            action: 't3.customHelpMenuSection.done',
+            orgId,
+            fullName,
+        });
+    }
